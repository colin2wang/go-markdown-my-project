```ts
// scroll-spy：滚动到哪个分区就高亮哪个 tab
const pane = ref<HTMLElement | null>(null);
const activeTab = ref('basic');
function spy() {
  const el = pane.value; if (!el) return;
  const top = el.scrollTop;
  for (const id of ['basic', 'scope', 'exclude', 'include', 'export', 'redact', 'preview']) {
    const sec = el.querySelector(`#${id}`) as HTMLElement | null;
    if (sec && sec.offsetTop - el.offsetTop - 60 <= top) activeTab.value = id;
  }
}
```

```ts
// store 里 diff 出变更行号（简单做法：保存上次序列化文本做逐行对比）
function diffLines(prev: string, next: string): Set<number> {
  const a = prev.split('\n'), b = next.split('\n');
  const changed = new Set<number>();
  for (let i = 0; i < Math.max(a.length, b.length); i++) {
    if (a[i] !== b[i]) changed.add(i + 1);
  }
  return changed;
}
```
